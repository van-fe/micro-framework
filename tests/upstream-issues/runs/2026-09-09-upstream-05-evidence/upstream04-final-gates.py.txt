import subprocess,os,json,pathlib,datetime,sys,hashlib
root=pathlib.Path('/Users/fanfangbo/workspace/micro-framework')
out=root/'tests/upstream-issues/runs/2026-09-09-upstream-05-evidence'
phases=[
 ('architecture',['bun','run','check:architecture'],{}),
 ('typecheck',['bun','run','typecheck'],{}),
 ('build',['bun','run','build'],{}),
 ('unit',['bun','run','test:unit','--reporter=verbose','--reporter=json','--outputFile='+str(out/'final-unit.json')],{}),
 ('browser',['bun','run','test:browser','--reporter=verbose','--reporter=json','--outputFile='+str(out/'final-browser.json')],{}),
 ('e2e',['bunx','playwright','test','--max-failures=1','--reporter=line,json'],{'PLAYWRIGHT_JSON_OUTPUT_NAME':str(out/'final-e2e.json')}),
 ('production',['bunx','playwright','test','--config','playwright.production.config.ts','--reporter=line,json'],{'PLAYWRIGHT_JSON_OUTPUT_NAME':str(out/'final-production.json')}),
 ('artifacts',['bun','run','check:artifacts'],{})
]
records=[]
def source_snapshot():
 paths=set()
 for pattern in ['packages/*/src/**/*','examples/*/src/**/*','tests/e2e/**/*','tests/browser/**/*','tests/production/**/*','packages/*/package.json','examples/*/package.json','*.config.ts','package.json','bun.lock']:
  paths.update(p for p in root.glob(pattern) if p.is_file() and '__screenshots__' not in p.parts and (p.suffix in ['.ts','.tsx','.js','.mjs','.cjs','.json'] or p.name=='bun.lock'))
 return {str(p.relative_to(root)):hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted(paths)}
snapshot=source_snapshot()
(out/'final-source.json').write_text(json.dumps(snapshot,indent=2)+'\n')
for name,args,extra in phases:
 started=datetime.datetime.now(datetime.timezone.utc).isoformat();print('START',name,flush=True)
 log=out/('final-'+name+'.log')
 with log.open('w') as f:r=subprocess.run(args,cwd=root,env={**os.environ,**extra},stdout=f,stderr=subprocess.STDOUT)
 records.append({'phase':name,'argv':args,'env':extra,'startedAt':started,'finishedAt':datetime.datetime.now(datetime.timezone.utc).isoformat(),'exitCode':r.returncode,'log':str(log.relative_to(root))})
 (out/'final-commands.json').write_text(json.dumps(records,indent=2)+'\n')
 print('END',name,'exit',r.returncode,flush=True)
 if r.returncode and name!='artifacts':sys.exit(r.returncode)
assert snapshot==source_snapshot(),'Source changed during final gates'
print('Frozen source verified. Final command records saved; artifact gate remains separately reported.',flush=True)
